45const char *PowerManager::wakeupCauseString() const {
46 switch (_cause) {
47 case ESP_SLEEP_WAKEUP_TIMER: return "timer (periodic wake-up)";
48 case ESP_SLEEP_WAKEUP_EXT0: return "external signal (EXT0)";
49 case ESP_SLEEP_WAKEUP_EXT1:
50 if (wokeFromRain() && wokeFromButton()) return "rain gauge + reset button";
51 if (wokeFromRain()) return "rain gauge pulse (EXT1)";
52 if (wokeFromButton()) return "factory reset button (EXT1)";
53 return "external signal (EXT1)";
54 case ESP_SLEEP_WAKEUP_ULP: return "ULP coprocessor";
55 case ESP_SLEEP_WAKEUP_UNDEFINED:
56 default: return "power-on / reset (cold boot)";
57 }
58}

60void PowerManager::deepSleep(uint32_t seconds) {
61 esp_sleep_enable_timer_wakeup((uint64_t)seconds * 1000000ULL);
62
63 // Two wake sources share EXT1 (EXT0 only supports a single pin): a
64 // bucket tip on the rain gauge and the factory reset button, both
65 // pulling their pin low. The RTC domain needs its own pull-ups because
66 // the digital GPIO ones are powered down in deep sleep.
67 rtc_gpio_pullup_en(RAIN_GAUGE_PIN);
68 rtc_gpio_pulldown_dis(RAIN_GAUGE_PIN);
69 rtc_gpio_pullup_en(FACTORY_RESET_PIN);
70 rtc_gpio_pulldown_dis(FACTORY_RESET_PIN);
71 esp_sleep_enable_ext1_wakeup((1ULL << RAIN_GAUGE_PIN) | (1ULL << FACTORY_RESET_PIN),
72 ESP_EXT1_WAKEUP_ANY_LOW);
73
74 // Freeze the sensor rail control so it keeps its level while the CPU
75 // is powered down.
76 gpio_hold_en(SENSOR_POWER_PIN);
77 gpio_deep_sleep_hold_en();
78
79 Serial.flush();
80 esp_deep_sleep_start();
81
82 // esp_deep_sleep_start() does not return; satisfy [[noreturn]] anyway.
83 while (true) {}
84}

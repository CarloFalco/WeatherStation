21/**
22 * @brief Check that an active-low wake input is idle (high) before sleeping.
23 *
24 * EXT1 in ANY_LOW mode is level-triggered: a pin still low when the chip
25 * goes to sleep wakes it again immediately. That happens for real in two
26 * ways: the rain gauge reed still closed at the end of a quick-path wake
27 * (the same tip would be counted twice), and an input stuck low for good
28 * (bucket resting on the reed, water in an outdoor connector, button
29 * jammed) — which would turn deep sleep into a wake loop draining the
30 * battery in hours.
31 *
32 * Returns immediately when the pin is already high (the normal case, no
33 * cost); otherwise waits for a stable release up to a timeout.
34 *
35 * @param pin Active-low wake input.
36 * @return true if the pin is idle and can be armed as a wake source.
37 */
38static bool wakePinReleased(gpio_num_t pin) {
39 pinMode(pin, INPUT_PULLUP);
40 if (digitalRead(pin) == HIGH) {
41 return true;
42 }
43
44 uint32_t start = millis();
45 uint32_t highSince = 0;
46 bool high = false;
47 while (millis() - start < kWakePinReleaseTimeoutMs) {
48 if (digitalRead(pin) == HIGH) {
49 if (!high) {
50 high = true;
51 highSince = millis();
52 } else if (millis() - highSince >= kWakePinStableMs) {
53 return true;
54 }
55 } else {
56 high = false;
57 }
58 delay(1);
59 }
60 return false;
61}

92const char *PowerManager::wakeupCauseString() const {
93 switch (_cause) {
94 case ESP_SLEEP_WAKEUP_TIMER: return "timer (periodic wake-up)";
95 case ESP_SLEEP_WAKEUP_EXT0: return "external signal (EXT0)";
96 case ESP_SLEEP_WAKEUP_EXT1:
97 if (wokeFromRain() && wokeFromButton()) return "rain gauge + reset button";
98 if (wokeFromRain()) return "rain gauge pulse (EXT1)";
99 if (wokeFromButton()) return "factory reset button (EXT1)";
100 return "external signal (EXT1)";
101 case ESP_SLEEP_WAKEUP_ULP: return "ULP coprocessor";
102 case ESP_SLEEP_WAKEUP_UNDEFINED:
103 default: return "power-on / reset (cold boot)";
104 }
105}

107void PowerManager::deepSleep(uint32_t seconds) {
108 esp_sleep_enable_timer_wakeup((uint64_t)seconds * 1000000ULL);
109
110 // Two wake sources share EXT1 (EXT0 only supports a single pin): a
111 // bucket tip on the rain gauge and the factory reset button, both
112 // pulling their pin low. An input that is not released is left out for
113 // this cycle (the timer still wakes the station): see wakePinReleased().
114 uint64_t wakeMask = 0;
115 if (wakePinReleased(RAIN_GAUGE_PIN)) {
116 wakeMask |= 1ULL << RAIN_GAUGE_PIN;
117 } else {
118 log_w("Rain gauge input stuck low: wake-on-rain disabled for this sleep");
119 }
120 if (wakePinReleased(FACTORY_RESET_PIN)) {
121 wakeMask |= 1ULL << FACTORY_RESET_PIN;
122 } else {
123 log_w("Reset button input stuck low: button wake disabled for this sleep");
124 }
125
126 if (wakeMask != 0) {
127 // The RTC domain needs its own pull-ups: the digital GPIO ones are
128 // powered down in deep sleep (the board also has external ones).
129 rtc_gpio_pullup_en(RAIN_GAUGE_PIN);
130 rtc_gpio_pulldown_dis(RAIN_GAUGE_PIN);
131 rtc_gpio_pullup_en(FACTORY_RESET_PIN);
132 rtc_gpio_pulldown_dis(FACTORY_RESET_PIN);
133 esp_sleep_enable_ext1_wakeup(wakeMask, ESP_EXT1_WAKEUP_ANY_LOW);
134 }
135
136 // Freeze the sensor rail control so it keeps its level while the CPU
137 // is powered down.
138 gpio_hold_en(SENSOR_POWER_PIN);
139 gpio_deep_sleep_hold_en();
140
141 Serial.flush();
142 esp_deep_sleep_start();
143
144 // esp_deep_sleep_start() does not return; satisfy [[noreturn]] anyway.
145 while (true) {}
146}

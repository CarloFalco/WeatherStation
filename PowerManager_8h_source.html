1/**
2 * @file PowerManager.h
3 * @brief Deep-sleep orchestration and wake-up cause handling.
4 *
5 * Owns the station duty cycle: classifies why the MCU woke up, keeps the
6 * boot counter in RTC RAM up to date and puts the system back into deep
7 * sleep with the timer (and, from Increment 4, the rain-gauge EXT wake-up)
8 * armed.
9 *
10 * Usage example:
11 * @code
12 * PowerManager power;
13 * power.begin(); // classify wake-up, ++bootCount
14 * Serial.println(power.wakeupCauseString());
15 * power.deepSleep(config.station.wakeIntervalS); // never returns
16 * @endcode
17 */
18

27/**
28 * @brief Manages deep sleep, wake-up sources and the RTC-resident state.
29 */
30class PowerManager {
31public:
32 /**
33 * @brief Classify the wake-up cause and update the RTC state.
34 *
35 * Must be called once, early in setup(). Increments
36 * RtcState::bootCount (RTC RAM is zeroed by hardware on power-on,
37 * so the first boot reads 1).
38 */
39 void begin();
40
41 /** @return Raw ESP-IDF wake-up cause of the current boot. */
42 esp_sleep_wakeup_cause_t wakeupCause() const { return _cause; }
43
44 /** @return Human-readable description of the wake-up cause. */
45 const char *wakeupCauseString() const;
46
47 /** @return true on power-on/reset, false when waking from deep sleep. */
48 bool isColdBoot() const { return _cause == ESP_SLEEP_WAKEUP_UNDEFINED; }
49
50 /** @return true if a rain-gauge pulse woke the station. */
51 bool wokeFromRain() const { return _extPins & (1ULL << RAIN_GAUGE_PIN); }
52
53 /** @return true if the factory-reset button woke the station. */
54 bool wokeFromButton() const { return _extPins & (1ULL << FACTORY_RESET_PIN); }
55
56 /**
57 * @brief Drive the sensor rail load switch (AS5600 + soil probe).
58 *
59 * Hardware: GPIO high -> NPN -> IRF9540 P-MOSFET gate low -> rail on.
60 * A 10k pull-down keeps the rail off whenever the pin is not driven,
61 * so an unprogrammed or resetting MCU leaves the sensors unpowered.
62 * The level is retained during deep sleep (RTC hold).
63 *
64 * @param on true to power the sensor rail.
65 */
66 void setSensorRail(bool on);
67
68 /**
69 * @brief Block until the sensor rail has been up for @p settleMs.
70 *
71 * Only waits for the time not already spent booting, so on a release
72 * build (no USB wait) the sensors still get their full settling time
73 * while a debug build usually waits for nothing.
74 *
75 * @param settleMs Settling time required after power-up [ms].
76 */
77 void waitSensorRailSettled(uint16_t settleMs) const;
78
79 /**
80 * @brief Enter deep sleep with the timer wake-up armed.
81 *
82 * Flushes the serial output before powering down. Execution restarts
83 * from setup() at the next wake-up: this function never returns.
84 *
85 * @param seconds Sleep duration [s] (typically station.wakeIntervalS).
86 */
87 [[noreturn]] void deepSleep(uint32_t seconds);
88
89private:
90 esp_sleep_wakeup_cause_t _cause = ESP_SLEEP_WAKEUP_UNDEFINED;
91 uint64_t _extPins = 0; ///< Bitmask of the EXT1 pins that caused the wake-up.
92 uint32_t _railOnMs = 0; ///< millis() when the sensor rail was switched on.
93};

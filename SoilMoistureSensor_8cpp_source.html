11bool SoilMoistureSensor::begin() {
12 // 12 dB attenuation = full 0-3.1 V input span (the probe swings close
13 // to the rail when dry). 12-bit resolution is the Arduino default.
14 analogSetPinAttenuation(SOIL_MOISTURE_PIN, ADC_11db);

18bool SoilMoistureSensor::read(JsonObject &out) {
19 uint32_t sum = 0;
20 for (uint8_t i = 0; i < _samples; i++) {
21 sum += analogRead(SOIL_MOISTURE_PIN);
22 delay(2); // let the ADC sample-and-hold settle between readings
23 }
24 uint16_t raw = (uint16_t)(sum / _samples);
25
26 int percent = logic::soilRawToPercent(raw, _dryRaw, _wetRaw);
27 log_d("soil: raw %u (dry %u, wet %u) -> %d %%", raw, _dryRaw, _wetRaw, percent);
28
29 if (raw < kPlausibleMin || raw > kPlausibleMax) {
30 log_w("SoilMoisture: raw %u out of range, probe disconnected?", raw);
31 return false;
32 }
33 if (percent == logic::kSoilInvalid) {
34 log_w("SoilMoisture: invalid calibration (dry_raw must exceed wet_raw)");
35 return false;
36 }
37
38 out["soil"] = percent;
39 return true;
40}

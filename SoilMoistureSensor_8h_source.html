1/**
2 * @file SoilMoistureSensor.h
3 * @brief Capacitive soil moisture probe on an analog input.
4 *
5 * Contributes the "soil" field: soil moisture in percent, 0 = bone dry,
6 * 100 = saturated. The raw-to-percent conversion lives in
7 * src/logic/soil_moisture.* (pure logic, native-tested); this class only
8 * does the I/O and the plausibility check.
9 *
10 * Calibration comes from config.ini ([soil] dry_raw / wet_raw): measure
11 * the probe in air and in a glass of water, the raw counts are printed at
12 * debug level on every cycle.
13 */
14

20/**
21 * @brief ISensor implementation for the capacitive soil moisture probe.
22 */
23class SoilMoistureSensor : public ISensor {
24public:
25 const char *name() const override { return "SoilMoisture"; }
26
27 /**
28 * @brief Apply the calibration before begin().
29 * @param dryRaw ADC counts with the probe in air.
30 * @param wetRaw ADC counts with the probe in water.
31 * @param samples Readings averaged per measurement.
32 */
33 void configure(uint16_t dryRaw, uint16_t wetRaw, uint8_t samples) {
34 _dryRaw = dryRaw;
35 _wetRaw = wetRaw;
36 _samples = samples;
37 }
38
39 /**
40 * @brief Configure the ADC input (12 dB attenuation, full 0-3.1 V span).
41 * @return Always true: a plain analog input cannot be probed.
42 */
43 bool begin() override;
44
45 /**
46 * @brief Average the ADC and publish "soil" [%].
47 * @param out Root JSON object of the telemetry message.
48 * @return false if the probe looks disconnected or the calibration is
49 * invalid — the field is then absent, never invented.
50 */
51 bool read(JsonObject &out) override;
52
53private:
54 /// A floating input rails to one extreme: readings outside this window
55 /// mean no probe connected (or a wiring fault), not dry/wet soil.
56 static constexpr uint16_t kPlausibleMin = 50;
57 static constexpr uint16_t kPlausibleMax = 4045;
58
59 uint16_t _dryRaw = 3000; ///< Calibration in air, from config.
60 uint16_t _wetRaw = 1300; ///< Calibration in water, from config.
61 uint8_t _samples = 8; ///< Readings averaged per measurement.
62};
